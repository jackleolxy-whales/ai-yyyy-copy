        async function replaceInvalidEdlSources() {
            try {
                const res = await fetch('/uploads/list');
                const j = await res.json();
                if (!j.success) { console.warn('list uploads failed', j.error); return false; }
                const files = j.files || [];
                const fallback = files.find(f=>f.endsWith('_11_compressed.mp4')) || files.find(f=>f.toLowerCase().endsWith('.mp4'));
                if (!fallback) { console.warn('no fallback file'); return false; }
                const tbody = document.querySelector('#edlTable tbody');
                const rows = tbody ? tbody.querySelectorAll('tr') : [];
                let replaced = 0;
                for (const r of rows) {
                    const inputs = r.querySelectorAll('input');
                    const source = (inputs[1]?.value || '').trim();
                    if (!source) continue;
                    const check = await fetch('/video/edl/resolve', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ name: source }) }).then(r=>r.json()).catch(()=>({ success:false }));
                    if (!check.success) { inputs[1].value = fallback; replaced++; }
                }
                console.log('EDL sources replaced', { replaced, fallback });
                return replaced > 0;
            } catch(e) { console.error('replaceInvalidEdlSources error', e); return false; }
        }
